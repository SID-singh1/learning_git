git init - intialize a new git repository
git status - to check status
git add filenames - pushes to tracked files, . to push all files in staging area

git commit - to save snapshot
added READM

git switch/checkout
git branch -a
git log - to view history, q to leave
    Space — next page
    b — previous page
    ↑ / ↓ — scroll line by line
    q — quit the view

git log --oneline - more compact versiom
